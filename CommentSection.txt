import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  MessageCircle,
  Send,
  ThumbsDown,
  ThumbsUp,
  MoreHorizontal,
  Pencil,
  Trash2,
  CornerUpLeft,
  Flag,
  Languages,
  Check,
  Loader2,
  X,
} from "lucide-react";

import { getCurrentUser } from "@/lib/auth";

import {
  canEditComment,
  type VideoComment,
  type CommentReportReason,
} from "@/lib/commentStorage";

import {
  checkRateLimit,
  isDuplicateComment,
  requiresCaptcha,
  recordCommentPost,
  validateCommentText,
} from "@/lib/commentModeration";

import {
  fetchComments,
  createBackendComment,
  updateBackendComment,
  deleteBackendComment,
  reactToBackendComment,
  reportBackendComment,
  type BackendComment,
} from "@/lib/commentApi";

type CommentSectionProps = {
  videoId: string;
};

type SortOption =
  | "newest"
  | "oldest"
  | "liked"
  | "relevant";

type CaptchaData = {
  first: number;
  second: number;
};

const languages = [
  "English",
  "Hindi",
  "Spanish",
  "French",
  "German",
];

const reportReasons: {
  label: string;
  value: CommentReportReason;
}[] = [
  {
    label: "Spam",
    value: "spam",
  },
  {
    label: "Harassment",
    value: "harassment",
  },
  {
    label: "Offensive content",
    value: "offensive",
  },
];

const CAPTCHA_VALIDITY = 5 * 60 * 1000;

function createCaptchaData(): CaptchaData {
  return {
    first:
      Math.floor(Math.random() * 9) + 1,
    second:
      Math.floor(Math.random() * 9) + 1,
  };
}

function mapBackendCommentToFrontend(
  comment: BackendComment
): VideoComment {
  const user =
    typeof comment.userId === "object"
      ? comment.userId
      : undefined;

  return {
    id: comment._id,
    videoId: comment.videoId as unknown as number,

    user: {
      username:
        comment.username ||
        user?.email?.split("@")[0] ||
        "user",

      displayName:
        comment.displayName ||
        user?.name ||
        "User",

      avatar:
        comment.avatar ||
        user?.avatar ||
        "",

      location:
        comment.location ||
        user?.location ||
        "India",

      email:
        user?.email || "",
    },

    text: comment.text,

    language:
      comment.language || "English",

    createdAt:
      comment.createdAt,

    updatedAt:
      comment.updatedAt,

    edited:
      Boolean(comment.edited),

    likes:
      Number(comment.likes || 0),

    dislikes:
      Number(comment.dislikes || 0),

    likedByCurrentUser:
      false,

    dislikedByCurrentUser:
      false,

    likedBy:
      comment.likedBy || [],

    dislikedBy:
      comment.dislikedBy || [],

    parentId:
      comment.parentId,

    replyCount:
      Number(comment.replyCount || 0),

    reported:
      Boolean(comment.reported),

    reportReasons: (comment.reportReasons || []).filter(
  (reason): reason is CommentReportReason =>
    reason === "spam" ||
    reason === "harassment" ||
    reason === "offensive"
),
  };
}

export default function CommentSection({
  videoId,
}: CommentSectionProps) {
  const [comments, setComments] =
    useState<VideoComment[]>([]);

  const [commentText, setCommentText] =
    useState("");

  const [commentLanguage, setCommentLanguage] =
    useState("English");

  const [replyingTo, setReplyingTo] =
    useState<string | null>(null);

  const [replyText, setReplyText] =
    useState("");

  const [editingId, setEditingId] =
    useState<string | null>(null);

  const [editText, setEditText] =
    useState("");

  const [sortOption, setSortOption] =
    useState<SortOption>("newest");

  const [error, setError] =
    useState("");

  const [success, setSuccess] =
    useState("");

  const [loading, setLoading] =
    useState(true);

  const [actionLoading, setActionLoading] =
    useState<string | null>(null);

  const [reportingId, setReportingId] =
    useState<string | null>(null);

    const [openCommentMenuId, setOpenCommentMenuId] =
    useState<string | null>(null);

  const [translatedComments, setTranslatedComments] =
    useState<Record<string, string>>({});

  const [translatingId, setTranslatingId] =
    useState<string | null>(null);

  const [targetLanguage, setTargetLanguage] =
    useState("Hindi");

  const [captchaOpen, setCaptchaOpen] =
    useState(false);

  const [captchaFor, setCaptchaFor] =
    useState<"comment" | "reply">("comment");

  const [captcha, setCaptcha] =
    useState<CaptchaData>(
      createCaptchaData()
    );

  const [captchaAnswer, setCaptchaAnswer] =
    useState("");

  const [
    captchaVerifiedUntil,
    setCaptchaVerifiedUntil,
  ] = useState(0);

  const currentUser =
    getCurrentUser();

  const clearMessages = () => {
    setError("");
    setSuccess("");
  };

  /*
   * ============================
   * LOAD COMMENTS FROM MONGODB
   * ============================
   */

  const loadComments = async () => {
    try {
      setLoading(true);
      setError("");

      const backendComments =
        await fetchComments(videoId);

      const mappedComments =
        backendComments.map(
          mapBackendCommentToFrontend
        );

      setComments(mappedComments);
    } catch (error) {
      console.error(
        "Load comments error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to load comments."
      );
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    if (!videoId) {
      setComments([]);
      setLoading(false);
      return;
    }

    loadComments();
  }, [videoId]);

  /*
   * ============================
   * USER HELPERS
   * ============================
   */

  const getCurrentUsername = () => {
    if (!currentUser) {
      return "guest";
    }

    return currentUser.email
      .split("@")[0]
      .toLowerCase()
      .replace(
        /[^a-z0-9_]/g,
        ""
      );
  };

  const requireLogin = () => {
    if (currentUser) {
      return true;
    }

    if (
      typeof window !== "undefined"
    ) {
      const returnUrl =
        `${window.location.pathname}${window.location.search}`;

      window.location.href =
        `/login?returnUrl=${encodeURIComponent(
          returnUrl
        )}`;
    }

    return false;
  };

  const clearError = () => {
    setError("");
  };

  /*
   * ============================
   * SORTING
   * ============================
   */

  const rootComments =
    useMemo(() => {
      const roots =
        comments.filter(
          (comment) =>
            comment &&
            comment.parentId === null
        );

      const sorted = [...roots];

      if (
        sortOption === "newest"
      ) {
        sorted.sort(
          (a, b) =>
            new Date(
              b.createdAt
            ).getTime() -
            new Date(
              a.createdAt
            ).getTime()
        );
      }

      if (
        sortOption === "oldest"
      ) {
        sorted.sort(
          (a, b) =>
            new Date(
              a.createdAt
            ).getTime() -
            new Date(
              b.createdAt
            ).getTime()
        );
      }

      if (
        sortOption === "liked"
      ) {
        sorted.sort(
          (a, b) =>
            b.likes - a.likes
        );
      }

      if (
        sortOption === "relevant"
      ) {
        sorted.sort(
          (a, b) => {
            const aScore =
              a.likes * 3 +
              a.replyCount * 2;

            const bScore =
              b.likes * 3 +
              b.replyCount * 2;

            return (
              bScore - aScore
            );
          }
        );
      }

      return sorted;
    }, [
      comments,
      sortOption,
    ]);

  const getReplies = (
    parentId: string
  ) => {
    return comments
      .filter(
        (comment) =>
          comment.parentId ===
          parentId
      )
      .sort(
        (a, b) =>
          new Date(
            a.createdAt
          ).getTime() -
          new Date(
            b.createdAt
          ).getTime()
      );
  };

  /*
   * ============================
   * CAPTCHA
   * ============================
   */

  const createNewCaptcha = () => {
    setCaptcha(
      createCaptchaData()
    );

    setCaptchaAnswer("");
  };

  const openCaptcha = (
    type: "comment" | "reply"
  ) => {
    setCaptchaFor(type);
    createNewCaptcha();
    setCaptchaOpen(true);
    clearMessages();
  };

  const verifyCaptcha = () => {
    const expected =
      captcha.first +
      captcha.second;

    const answer =
      Number(captchaAnswer);

    if (
      answer !== expected
    ) {
      setError(
        "Incorrect CAPTCHA answer."
      );

      createNewCaptcha();

      return false;
    }

    setCaptchaOpen(false);
    setCaptchaAnswer("");
    setError("");

    setCaptchaVerifiedUntil(
      Date.now() +
        CAPTCHA_VALIDITY
    );

    return true;
  };

  const shouldRequireCaptcha =
    () => {
      if (
        Date.now() <
        captchaVerifiedUntil
      ) {
        return false;
      }

      return requiresCaptcha();
    };

  /*
   * ============================
   * VALIDATION
   * ============================
   */

  const validatePost = (
    text: string
  ) => {
    clearMessages();

    if (!currentUser) {
      setError(
        "Please sign in before posting a comment."
      );

      return false;
    }

    const validation =
      validateCommentText(text);

    if (!validation.valid) {
      setError(
        validation.reason
      );

      return false;
    }

    const rate =
      checkRateLimit();

    if (!rate.allowed) {
      setError(
        "You are posting too quickly. Please wait before posting another comment."
      );

      return false;
    }

    const currentActor =
      currentUser.email;

    const existingTexts =
      comments
        .filter(
          (comment) =>
            comment.user.email ===
            currentActor
        )
        .map(
          (comment) =>
            comment.text
        );

    if (
      isDuplicateComment(
        text,
        existingTexts
      )
    ) {
      setError(
        "You already posted a similar comment."
      );

      return false;
    }

    return true;
  };

  /*
   * ============================
   * CREATE COMMENT
   * ============================
   */

  const saveNewComment = async (
    text: string,
    parentId:
      | string
      | null,
    language: string
  ) => {
    try {
      setActionLoading(
        parentId
          ? `reply-${parentId}`
          : "new-comment"
      );

      const created =
        await createBackendComment(
          videoId,
          text,
          language,
          parentId
        );

      recordCommentPost();

      const mapped =
        mapBackendCommentToFrontend(
          created
        );

      setComments(
        (previous) => [
          ...previous,
          mapped,
        ]
      );

      return true;
    } catch (error) {
      console.error(
        "Create comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to save comment."
      );

      return false;
    } finally {
      setActionLoading(null);
    }
  };

  const handleAddComment = async () => {
    if (!requireLogin()) {
      return;
    }

    const text =
      commentText.trim();

    if (!validatePost(text)) {
      return;
    }

    if (
      shouldRequireCaptcha() &&
      !captchaOpen
    ) {
      openCaptcha("comment");
      return;
    }

    if (
      captchaOpen &&
      captchaFor === "comment"
    ) {
      if (!verifyCaptcha()) {
        return;
      }
    }

    const saved =
      await saveNewComment(
        text,
        null,
        commentLanguage
      );

    if (!saved) {
      return;
    }

    setCommentText("");

    setSuccess(
      "Comment posted successfully."
    );
  };

  /*
   * ============================
   * REPLY
   * ============================
   */

  const handleReply = async (
    parentComment: VideoComment
  ) => {
    if (!requireLogin()) {
      return;
    }

    const text =
      replyText.trim();

    if (!validatePost(text)) {
      return;
    }

    if (
      shouldRequireCaptcha() &&
      !captchaOpen
    ) {
      openCaptcha("reply");
      return;
    }

    if (
      captchaOpen &&
      captchaFor === "reply"
    ) {
      if (!verifyCaptcha()) {
        return;
      }
    }

    const saved =
      await saveNewComment(
        text,
        parentComment.id,
        commentLanguage
      );

    if (!saved) {
      return;
    }

    setReplyText("");
    setReplyingTo(null);

    setSuccess(
      "Reply posted successfully."
    );
  };

  const handleCaptchaSubmit =
    async () => {
      if (!captchaOpen) {
        return;
      }

      if (!verifyCaptcha()) {
        return;
      }

      if (
        captchaFor === "comment"
      ) {
        await handleAddComment();
        return;
      }

      if (
        captchaFor === "reply" &&
        replyingTo
      ) {
        const parent =
          comments.find(
            (comment) =>
              comment.id ===
              replyingTo
          );

        if (parent) {
          await handleReply(
            parent
          );
        }
      }
    };

  /*
   * ============================
   * OWN COMMENT
   * ============================
   */

  const isOwnComment = (
    comment: VideoComment
  ) => {
    if (!currentUser) {
      return false;
    }

    /*
     * Backend comments are created
     * using the authenticated JWT user.
     *
     * The frontend compares email
     * with the currently logged-in user.
     */

    return (
      comment.user.email ===
      currentUser.email
    );
  };

  /*
   * ============================
   * EDIT
   * ============================
   */

  const startEditing = (
    comment: VideoComment
  ) => {
    if (!isOwnComment(comment)) {
      setError(
        "You can only edit your own comment."
      );

      return;
    }

    if (!canEditComment(comment)) {
      setError(
        "The edit time limit for this comment has expired."
      );

      return;
    }

    setEditingId(
      comment.id
    );

    setEditText(
      comment.text
    );

    setReplyingTo(null);

    clearMessages();
  };

  const cancelEditing = () => {
    setEditingId(null);
    setEditText("");
  };

  const handleEdit = async (
    comment: VideoComment
  ) => {
    if (!isOwnComment(comment)) {
      setError(
        "You can only edit your own comment."
      );

      return;
    }

    const text =
      editText.trim();

    if (!text) {
      setError(
        "Comment cannot be empty."
      );

      return;
    }

    const validation =
      validateCommentText(text);

    if (!validation.valid) {
      setError(
        validation.reason
      );

      return;
    }

    try {
      setActionLoading(
        `edit-${comment.id}`
      );

      const updated =
        await updateBackendComment(
          comment.id,
          text,
          comment.language
        );

      const mapped =
        mapBackendCommentToFrontend(
          updated
        );

      setComments(
        (previous) =>
          previous.map(
            (item) =>
              item.id ===
              comment.id
                ? mapped
                : item
          )
      );

      cancelEditing();

      setSuccess(
        "Comment updated successfully."
      );
    } catch (error) {
      console.error(
        "Update comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to update comment."
      );
    } finally {
      setActionLoading(null);
    }
  };

  /*
   * ============================
   * DELETE
   * ============================
   */

  const handleDelete = async (
    comment: VideoComment
  ) => {
    if (!isOwnComment(comment)) {
      setError(
        "You can only delete your own comment."
      );

      return;
    }

    const confirmed =
      window.confirm(
        "Are you sure you want to delete this comment?"
      );

    if (!confirmed) {
      return;
    }

    try {
      setActionLoading(
        `delete-${comment.id}`
      );

      await deleteBackendComment(
        comment.id
      );

      setComments(
        (previous) =>
          previous.filter(
            (item) =>
              item.id !==
              comment.id
          )
      );

      if (
        editingId ===
        comment.id
      ) {
        cancelEditing();
      }

      if (
        replyingTo ===
        comment.id
      ) {
        setReplyingTo(null);
        setReplyText("");
      }

      setSuccess(
        "Comment deleted successfully."
      );
    } catch (error) {
      console.error(
        "Delete comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to delete comment."
      );
    } finally {
      setActionLoading(null);
    }
  };

  /*
   * ============================
   * LIKE
   * ============================
   */

  const handleLike = async (
    commentId: string
  ) => {
    if (!requireLogin()) {
      return;
    }

    try {
      setActionLoading(
        `like-${commentId}`
      );

      const updated =
        await reactToBackendComment(
          commentId,
          "like"
        );

      const mapped =
        mapBackendCommentToFrontend(
          updated
        );

      setComments(
        (previous) =>
          previous.map(
            (comment) =>
              comment.id ===
              commentId
                ? mapped
                : comment
          )
      );
    } catch (error) {
      console.error(
        "Like comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to update reaction."
      );
    } finally {
      setActionLoading(null);
    }
  };

  /*
   * ============================
   * DISLIKE
   * ============================
   */

  const handleDislike = async (
    commentId: string
  ) => {
    if (!requireLogin()) {
      return;
    }

    try {
      setActionLoading(
        `dislike-${commentId}`
      );

      const updated =
        await reactToBackendComment(
          commentId,
          "dislike"
        );

      const mapped =
        mapBackendCommentToFrontend(
          updated
        );

      setComments(
        (previous) =>
          previous.map(
            (comment) =>
              comment.id ===
              commentId
                ? mapped
                : comment
          )
      );
    } catch (error) {
      console.error(
        "Dislike comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to update reaction."
      );
    } finally {
      setActionLoading(null);
    }
  };

  /*
   * ============================
   * REPORT
   * ============================
   */

  const handleReport = async (
    comment: VideoComment,
    reason: CommentReportReason
  ) => {
    if (!requireLogin()) {
      return;
    }

    try {
      setActionLoading(
        `report-${comment.id}`
      );

      await reportBackendComment(
        comment.id,
        reason
      );

      setReportingId(null);

      setComments(
        (previous) =>
          previous.map(
            (item) =>
              item.id ===
              comment.id
                ? {
                    ...item,
                    reported: true,
                    reportReasons:
                      Array.from(
                        new Set([
                          ...(item.reportReasons ||
                            []),
                          reason,
                        ])
                      ),
                  }
                : item
          )
      );

      setSuccess(
        "Comment reported for review."
      );
    } catch (error) {
      console.error(
        "Report comment error:",
        error
      );

      setError(
        error instanceof Error
          ? error.message
          : "Unable to report this comment."
      );
    } finally {
      setActionLoading(null);
    }
  };

  /*
   * ============================
   * TRANSLATION
   * ============================
   */

  const handleTranslate = async (
    comment: VideoComment
  ) => {
    clearMessages();

    if (
      comment.language ===
      targetLanguage
    ) {
      setTranslatedComments(
        (previous) => ({
          ...previous,
          [comment.id]:
            comment.text,
        })
      );

      return;
    }

    try {
      setTranslatingId(
        comment.id
      );

      const response =
        await fetch(
          "/api/translate",
          {
            method: "POST",
            headers: {
              "Content-Type":
                "application/json",
            },
            body: JSON.stringify({
              text:
                comment.text,
              sourceLanguage:
                comment.language ||
                "English",
              targetLanguage,
            }),
          }
        );

      const data =
        await response.json();

      if (!response.ok) {
        throw new Error(
          data?.error ||
            "Translation failed."
        );
      }

      if (
        typeof data?.translatedText !==
        "string"
      ) {
        throw new Error(
          "Invalid translation response."
        );
      }

      setTranslatedComments(
        (previous) => ({
          ...previous,
          [comment.id]:
            data.translatedText,
        })
      );

      setSuccess(
        `Comment translated to ${targetLanguage}.`
      );
    } catch (error) {
      console.error(
        "Translation error:",
        error
      );

      setError(
        "Unable to translate this comment right now."
      );
    } finally {
      setTranslatingId(
        null
      );
    }
  };

  const clearTranslation = (
    commentId: string
  ) => {
    setTranslatedComments(
      (previous) => {
        const next = {
          ...previous,
        };

        delete next[
          commentId
        ];

        return next;
      }
    );
  };

  /*
   * ============================
   * UI HELPERS
   * ============================
   */

  const formatDate = (
    date: string
  ) => {
    return new Date(
      date
    ).toLocaleString(
      undefined,
      {
        dateStyle:
          "medium",
        timeStyle:
          "short",
      }
    );
  };

  const getInitials = (
    name: string
  ) => {
    const initials =
      name
        .split(" ")
        .filter(Boolean)
        .map(
          (word) =>
            word[0]
        )
        .join("")
        .slice(0, 2)
        .toUpperCase();

    return (
      initials || "U"
    );
  };

  const renderTextWithMentions = (
    text: string
  ) => {
    const parts =
      text.split(
        /(@[a-zA-Z0-9_]+)/g
      );

    return parts.map(
      (part, index) => {
        if (
          /^@[a-zA-Z0-9_]+$/.test(
            part
          )
        ) {
          return (
            <span
              key={`${part}-${index}`}
              className="font-medium text-blue-600"
            >
              {part}
            </span>
          );
        }

        return (
          <span
            key={`${part}-${index}`}
          >
            {part}
          </span>
        );
      }
    );
  };

  const getDisplayText = (
    comment: VideoComment
  ) => {
    return (
      translatedComments[
        comment.id
      ] || comment.text
    );
  };

  /*
   * ============================
   * RENDER COMMENT
   * ============================
   */

  const renderComment = (
    comment: VideoComment,
    isReply = false
  ) => {
    const replies =
      getReplies(
        comment.id
      );

    const ownComment =
      isOwnComment(comment);

    const currentAction =
      actionLoading;

    return (
      <article
        key={comment.id}
        className={`flex gap-3 ${
          isReply
            ? "ml-6 mt-5 sm:ml-12"
            : ""
        }`}
      >
        {/* Avatar */}
