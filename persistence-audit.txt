export type ChannelProfile = {
  name: string;
  username: string;
  description: string;
  avatar: string;
  banner: string;
};

const DB_NAME = "youtube_channel_database";
const STORE_NAME = "channel_profile";
const DB_VERSION = 1;

const DEFAULT_PROFILE: ChannelProfile = {
  name: "Abishek Dubey",
  username: "@abishekdubey",
  description:
    "Welcome to my channel. I share web development, React, Next.js, JavaScript, AI and technology content.",
  avatar: "",
  banner: "",
};

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof window === "undefined") {
      reject(
        new Error("IndexedDB is only available in the browser.")
      );
      return;
    }

    const request = indexedDB.open(
      DB_NAME,
      DB_VERSION
    );

    request.onupgradeneeded = () => {
      const db = request.result;

      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, {
          keyPath: "id",
        });
      }
    };

    request.onsuccess = () => {
      resolve(request.result);
    };

    request.onerror = () => {
      reject(
        request.error ||
          new Error("Unable to open channel database.")
      );
    };
  });
}

export async function getChannelProfile(): Promise<ChannelProfile> {
  if (typeof window === "undefined") {
    return DEFAULT_PROFILE;
  }

  try {
    const db = await openDatabase();

    return new Promise((resolve, reject) => {
      const transaction = db.transaction(
        STORE_NAME,
        "readonly"
      );

      const store =
        transaction.objectStore(STORE_NAME);

      const request = store.get("profile");

      request.onsuccess = () => {
        db.close();

        if (!request.result) {
          resolve(DEFAULT_PROFILE);
          return;
        }

        resolve({
          ...DEFAULT_PROFILE,
          ...request.result.data,
        });
      };

      request.onerror = () => {
        db.close();

        reject(
          request.error ||
            new Error(
              "Unable to read channel profile."
            )
        );
      };
    });
  } catch (error) {
    console.error(
      "Failed to load channel profile:",
      error
    );

    return DEFAULT_PROFILE;
  }
}

export async function saveChannelProfile(
  profile: ChannelProfile
): Promise<void> {
  const db = await openDatabase();

  return new Promise((resolve, reject) => {
    const transaction = db.transaction(
      STORE_NAME,
      "readwrite"
    );

    const store =
      transaction.objectStore(STORE_NAME);

    store.put({
      id: "profile",
      data: profile,
    });

    transaction.oncomplete = () => {
      db.close();
      resolve();
    };

    transaction.onerror = () => {
      db.close();

      reject(
        transaction.error ||
          new Error(
            "Unable to save channel profile."
          )
      );
    };
  });
}

export async function resetChannelProfile(): Promise<void> {
  const db = await openDatabase();

  return new Promise((resolve, reject) => {
    const transaction = db.transaction(
      STORE_NAME,
      "readwrite"
    );

    const store =
      transaction.objectStore(STORE_NAME);

    store.delete("profile");

    transaction.oncomplete = () => {
      db.close();
      resolve();
    };

    transaction.onerror = () => {
      db.close();

      reject(
        transaction.error ||
          new Error(
            "Unable to reset channel profile."
          )
      );
    };
  });
}

function getCurrentActorId() {
  if (typeof window === "undefined") {
    return "guest";
  }

  const user = getCurrentUser();

  if (!user) {
    return "guest";
  }

  return (
    user.email ||
    user.name ||
    "guest"
  ).toLowerCase();
}

function readStorage<T>(
  key: string,
  fallback: T
): T {
  if (typeof window === "undefined") {
    return fallback;
  }

  try {
    const data =
      localStorage.getItem(key);

    if (!data) {
      return fallback;
    }

    const parsed = JSON.parse(data);

    return parsed as T;
  } catch (error) {
    console.error(
      `Failed to read ${key}:`,
      error
    );

    return fallback;
  }
}

function writeStorage<T>(
  key: string,
  value: T
) {
  if (typeof window === "undefined") {
    return false;
  }

  try {
    localStorage.setItem(
      key,
      JSON.stringify(value)
    );

    return true;
  } catch (error) {
    console.error(
      `Failed to save ${key}:`,
      error
    );

    return false;
  }
}

function normalizeComment(
  comment: VideoComment
): VideoComment {
  const likedBy = Array.isArray(
    comment.likedBy
  )
    ? comment.likedBy
    : [];

  const dislikedBy = Array.isArray(
    comment.dislikedBy
  )
    ? comment.dislikedBy
    : [];

  const currentActor =
    getCurrentActorId();

  return {
    ...comment,

    user: {
      username:
        comment.user?.username ||
        "user",
      displayName:
        comment.user?.displayName ||
        "Guest User",
      avatar:
        comment.user?.avatar ||
        "",
      location:
        comment.user?.location ||
        "India",
      email:
        comment.user?.email ||
        "",
    },

    text:
      typeof comment.text ===
      "string"
        ? comment.text
        : "",

    language:
      comment.language ||
      "English",

    createdAt:
      comment.createdAt ||
      new Date().toISOString(),

    updatedAt:
      comment.updatedAt ||
      comment.createdAt ||
      new Date().toISOString(),

    edited:
      Boolean(comment.edited),

    likes:
      typeof comment.likes ===
      "number"
        ? Math.max(
            0,
            comment.likes
          )
        : 0,

    dislikes:
      typeof comment.dislikes ===
      "number"
        ? Math.max(
            0,
            comment.dislikes
          )
        : 0,

    likedByCurrentUser:
      likedBy.includes(
        currentActor
      ),

    dislikedByCurrentUser:
      dislikedBy.includes(
        currentActor
      ),

    likedBy,

    dislikedBy,

    parentId:
      typeof comment.parentId ===
      "string"
        ? comment.parentId
        : null,

    replyCount:
      typeof comment.replyCount ===
      "number"
        ? Math.max(
            0,
            comment.replyCount
          )
        : 0,

    reported:
      Boolean(comment.reported),

    reportReasons:
      Array.isArray(
        comment.reportReasons
      )
        ? comment.reportReasons
        : [],
  };
}

function getStoredComments(): VideoComment[] {

export function saveComment(
  comment: VideoComment
) {
  const comments =
    getStoredComments();

  const existing =
    comments.some(
      (item) =>
        item.id === comment.id
    );

  if (existing) {
    return false;
  }

  const normalizedComment =
    normalizeComment(comment);

  saveAllComments([
    normalizedComment,
    ...comments,
  ]);

  /*
   * Update parent's reply count.
   */
  if (
    normalizedComment.parentId
  ) {
    updateReplyCount(
      normalizedComment.parentId
    );
  }

  return true;
}

export function updateReplyCount(
  parentId: string
) {
  const comments =
    getStoredComments();

  const replyCount =
    comments.filter(
      (comment) =>
        comment.parentId ===
        parentId
    ).length;

  const updated =
    comments.map(
      (comment) =>
        comment.id === parentId
          ? {
              ...comment,
              replyCount,
            }
          : comment
    );

  saveAllComments(updated);
}

export function canEditComment(
  comment: VideoComment,
  actorId?: string
) {
  const currentActor =
    (
      actorId ||
      getCurrentActorId()
    ).toLowerCase();

  const commentOwner =
    (
      comment.user.email ||
      comment.user.username
    ).toLowerCase();

  const isOwner =
    currentActor ===
    commentOwner;

  if (!isOwner) {
    return false;
  }

  const createdAt =
    new Date(
      comment.createdAt
    ).getTime();

  const now = Date.now();

  return (
    now - createdAt <=
    EDIT_TIME_LIMIT
  );
}

export function getRemainingEditTime(
  comment: VideoComment
) {
  const createdAt =
    new Date(
      comment.createdAt
    ).getTime();

  const expiresAt =
    createdAt +
    EDIT_TIME_LIMIT;

  return Math.max(
    0,
    expiresAt - Date.now()
  );
}

export function updateComment(
  commentId: string,
  text: string,
  expectedUpdatedAt?: string
) {
  const comments =
    getStoredComments();

  const current =
    comments.find(
      (comment) =>
        comment.id ===
        commentId
    );

  if (!current) {
    return {
      success: false,
      reason:
        "Comment not found.",
    };
  }

  const actor =
    getCurrentActorId();

  const owner =
    (
      current.user.email ||
      current.user.username
    ).toLowerCase();

  if (
    actor !== owner
  ) {
    return {
      success: false,
      reason:
        "You can only edit your own comment.",
    };
  }

  if (
    expectedUpdatedAt &&
    current.updatedAt !==
      expectedUpdatedAt
  ) {
    return {
      success: false,
      reason:
        "This comment was already changed. Please refresh and try again.",
    };
  }

  if (
    !canEditComment(
      current,
      actor
    )
  ) {
    return {
      success: false,
      reason:
        "The edit time limit for this comment has expired.",
    };
  }

  const cleanText =
    text.trim();

  if (!cleanText) {
    return {
      success: false,
      reason:
        "Comment cannot be empty.",
    };
  }

  const updatedAt =
    new Date().toISOString();

  const updatedComments =
    comments.map(
      (comment) =>
        comment.id ===
        commentId
          ? {
              ...comment,
              text: cleanText,
              updatedAt,
              edited: true,
            }
          : comment
    );

  const saved =
    saveAllComments(
      updatedComments
    );

  if (!saved) {
    return {
      success: false,
      reason:
        "Unable to save comment.",
    };
  }

  return {
    success: true,
    reason:
      "Comment updated successfully.",
  };
}

export function deleteComment(

  const comments =

    comments.find(

        item.id === commentId
    );

  if (!comment) {

              ...item,
              parentId: null,
            }
          : item
      );

  const saved =
    saveAllComments(
      updatedComments
    );

  if (!saved) {
    return {
      success: false,
      reason:
        "Unable to delete comment.",
    };
  }

  return {
    success: true,
    reason:
      "Comment deleted successfully.",
  };

export function addCommentReport(
  report: CommentReport
) {
  const reports =
    readStorage<
      CommentReport[]
    >(
      REPORT_KEY,
      []
    );

  const alreadyReported =
    reports.some(
      (item) =>
        item.commentId ===
          report.commentId &&
        item.reportedBy ===
          report.reportedBy &&
        item.reason ===
          report.reason
    );

  if (
    alreadyReported
  ) {
    return false;
  }

  const comment =
    getCommentById(
      report.commentId
    );

  if (!comment) {
    return false;
  }

  const updatedReport =
    {
      ...report,
      reportedBy:
        report.reportedBy ||
        getCurrentActorId(),
    };

  const saved =
    writeStorage(
      REPORT_KEY,
      [
        updatedReport,
        ...reports,
      ]
    );

  if (!saved) {
    return false;
  }

  /*
   * Mark the comment as reported.
   */
  const comments =
    getStoredComments();

  const updatedComments =
    comments.map(
      (item) =>
        item.id ===
        report.commentId
          ? {
              ...item,
              reported: true,
              reportReasons:
                item.reportReasons.includes(
                  report.reason
                )
                  ? item.reportReasons
                  : [
                      ...item.reportReasons,
                      report.reason,
                    ],
            }
          : item
    );

  saveAllComments(
    updatedComments
  );

  return true;
}

export function getDeviceId() {
  if (
    typeof window ===
    "undefined"
  ) {
    return "server";
  }

  const raw = [
    navigator.userAgent,
    navigator.language,
    navigator.platform,
    window.screen.width,
    window.screen.height,
    window.devicePixelRatio,
    Intl.DateTimeFormat()
      .resolvedOptions()
      .timeZone,
  ].join("|");

  let hash = 0;

  for (
    let index = 0;
    index < raw.length;
    index++
  ) {
    hash =
      (hash << 5) -
      hash +
      raw.charCodeAt(index);

    hash |= 0;
  }

  return `device_${Math.abs(
    hash
  )}`;
}

export function getBrowserName() {
  if (
    typeof navigator ===
    "undefined"
  ) {
    return "Unknown";
  }

  const userAgent =
    navigator.userAgent;

  if (
    /Edg/i.test(
      userAgent
    )
  ) {
    return "Microsoft Edge";
  }

  if (
    /Chrome/i.test(
      userAgent
    ) &&
    !/Edg/i.test(
      userAgent
    )
  ) {
    return "Google Chrome";
  }

  if (
    /Firefox/i.test(
      userAgent
    )
  ) {
    return "Mozilla Firefox";
  }

  if (
    /Safari/i.test(
      userAgent
    ) &&
    !/Chrome/i.test(
      userAgent
    )
  ) {
    return "Safari";
  }

  return "Unknown Browser";
}

export function getOperatingSystem() {

    "undefined"
  ) {
    return "Unknown";
  }

  const userAgent =
    navigator.userAgent;

  if (
    /Windows/i.test(
      userAgent
    )
  ) {
    return "Windows";
  }

  if (

      userAgent
    )
  ) {
    return "macOS";
  }

  if (
    /Android/i.test(
      userAgent
    )

export function isDeviceTrustedForLogin(
  userEmail: string,
  deviceId: string,
  ipAddress: string,
  location: string
) {
  const device =
    getStoredDevices().find(
      (item) =>
        item.userEmail ===

        item.id ===
          deviceId
    );

  if (!device) {

  }

  if (!device.trusted) {
    return false;
  }

  /*
   * When an old record did not contain
   * IP/location, require verification.
   */
  if (
    !device.ipAddress ||
    device.ipAddress ===
      "Unknown"
  ) {
    return false;
  }

  /*
   * Unknown incoming IP/location should
   * not automatically bypass verification.

  if (

    return false;

  if (

    location === "Unknown"
  ) {
    return false;
  }

  const sameIp =
    device.ipAddress ===
    ipAddress;

  const sameLocation =
    device.location ===
    location;

  return (
    sameIp &&
    sameLocation
  );
}

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof window === "undefined") {
      reject(new Error("IndexedDB is only available in the browser."));
      return;
    }

    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onerror = () => {
      reject(request.error);
    };

    request.onupgradeneeded = () => {
      const database = request.result;

      if (!database.objectStoreNames.contains(STORE_NAME)) {
        database.createObjectStore(STORE_NAME, {
          keyPath: "downloadId",
        });
      }
    };

    request.onsuccess = () => {
      resolve(request.result);
    };
  });
}

export async function saveDownloadedVideo(
  downloadId: string,
  blob: Blob,

): Promise<void> {

  return new Promise((resolve, reject) => {

      STORE_NAME,

    const store = transaction.objectStore(STORE_NAME);

    const video: StoredVideo = {
      downloadId,
      blob,
      fileName,

    store.put(video);

    transaction.oncomplete = () => {
      database.close();
      resolve();
    };

    transaction.onerror = () => {
      database.close();
      reject(transaction.error);
    };
  });
}

          getRecordTime(
            existing

          DUPLICATE_WINDOW_MS

      data.videoId
    );

export function formatFileSize(
  bytes: number
) {
  if (
    !Number.isFinite(
      bytes
    ) ||
    bytes <= 0
  ) {
    return "0 B";
  }

  const units = [
    "B",
    "KB",
    "MB",
    "GB",
  ];

  const index =
    Math.min(
      units.length - 1,
      Math.floor(
        Math.log(bytes) /
          Math.log(1024)
      )
    );

  const size =
    bytes /
    Math.pow(
      1024,
      index
    );

  return `${size.toFixed(
    index === 0 ? 0 : 2
  )} ${units[index]}`;
}

  return "dark";

): ResolvedTheme {

    mode === "light"
  ) {
    return "light";
  }

  if (
    mode === "dark"
  ) {
    return "dark";
  }

  return getAutoTheme();
}

========================================= */

export function applyTheme(
  mode: ThemeMode
) {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  const theme =
    resolveTheme(mode);

  const html =
    document.documentElement;

  html.classList.remove(
    "theme-light",
    "theme-dark"
  );

  html.classList.add(
    theme === "dark"
      ? "theme-dark"
      : "theme-light"
  );

  html.setAttribute(
    "data-theme",
    theme
  );

  html.style.colorScheme =
    theme;

  /*
   * Useful for components that
   * need to react to a theme change.
   */
  window.dispatchEvent(
    new CustomEvent(
      "youtube-theme-updated",
      {
        detail: {
          mode,
          theme,
        },
      }
    )
  );
}

  description: string;
  category: string;
  visibility: string;

const DB_NAME = "youtube_clone_database";
const STORE_NAME = "uploaded_videos";
const DB_VERSION = 1;

function openDatabase(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    if (typeof window === "undefined") {
      reject(new Error("IndexedDB is only available in the browser."));
      return;
    }

    const request = indexedDB.open(DB_NAME, DB_VERSION);

    request.onupgradeneeded = () => {
      const db = request.result;

      if (!db.objectStoreNames.contains(STORE_NAME)) {
        db.createObjectStore(STORE_NAME, {
          keyPath: "id",
        });
      }
    };

    request.onsuccess = () => {
      resolve(request.result);
    };

    request.onerror = () => {
      reject(request.error);
    };
  });
}

export async function saveUploadedVideo(
  video: UploadedVideo
): Promise<void> {
  const db = await openDatabase();

  return new Promise((resolve, reject) => {
    const transaction = db.transaction(
      STORE_NAME,
      "readwrite"
    );

    const store = transaction.objectStore(STORE_NAME);

    store.put(video);

    transaction.oncomplete = () => {
      db.close();
      resolve();
    };

    transaction.onerror = () => {
      db.close();
      reject(transaction.error);
    };
  });
}

export async function getUploadedVideos(): Promise<
  UploadedVideo[]
> {
  const db = await openDatabase();

  return new Promise((resolve, reject) => {
    const transaction = db.transaction(
      STORE_NAME,
      "readonly"
    );

    const store = transaction.objectStore(STORE_NAME);
    const request = store.getAll();

    request.onsuccess = () => {
      db.close();

      const videos = request.result as UploadedVideo[];

      videos.sort((a, b) => b.id - a.id);

      resolve(videos);
    };

    request.onerror = () => {
      db.close();
      reject(request.error);
    };
  });
}

export async function deleteUploadedVideo(
  id: number
): Promise<void> {
  const db = await openDatabase();

  return new Promise((resolve, reject) => {
    const transaction = db.transaction(
      STORE_NAME,
      "readwrite"
    );

    const store = transaction.objectStore(STORE_NAME);

    store.delete(id);

    transaction.oncomplete = () => {
      db.close();
      resolve();
    };

    transaction.onerror = () => {
      db.close();
      reject(transaction.error);
    };
  });
}
